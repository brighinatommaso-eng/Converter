# Converter — dependencies (pip-installable)
# Install the core set with:  pip install -r requirements.txt

# --- Core (required) ---
PySide6>=6.6          # GUI (LGPL — distribution-safe Qt binding)
Pillow>=10.2          # image conversions
pandas>=2.1           # data / table conversions
openpyxl>=3.1         # read/write XLSX (used by pandas)
odfpy>=1.4            # read/write ODS  (used by pandas)
PyYAML>=6.0           # YAML data conversions
defusedxml>=0.7       # safe XML parsing (blocks billion-laughs / XXE on untrusted files)
pdfminer.six>=20221105 # PDF layout extraction for the structured document engine (pure-Python, MIT)
python-docx>=1.1      # read/write DOCX for the structured engine (MIT; lxml comes with it)
striprtf>=0.0.26      # rtf→txt (pure-Python, MIT) — works in Portable
trimesh>=4.0          # 3D mesh conversions (pure-Python, MIT; hard dep numpy comes with pandas)
markdown-it-py>=3.0   # Markdown reader of the markup engine (CommonMark/GFM, pure-Python, MIT)
onnxruntime>=1.17     # runs the local AI layout model for PDF reading (MIT, CPU only)
pypdfium2>=4.20       # renders PDF pages for the layout model (Apache-2.0/BSD-3, PDFium)
# The layout model itself (PP-DocLayoutV2, Apache-2.0, ~214 MB) is not on PyPI:
# the release build downloads it into models/ (see .github/workflows/build.yml).
# Without it PDF reading falls back to the geometric engine and says so.

# --- Optional features (install only if you need them) ---
# OCR of scanned PDFs (Full mode). ocrmypdf is MPL-2.0 and also needs the external
# Tesseract + Ghostscript installed on the system:
#     pip install ocrmypdf>=16.0
# 3MF import for the 3D category (trimesh reads 3MF only when lxml is present):
#     pip install lxml>=5
# HEIC/AVIF image codecs (native deps; decode works, HEIC encode may be limited):
#     pip install pillow-heif>=0.16
#     pip install pillow-avif-plugin>=1.4

# --- Non-pip system tools (NOT installed by pip) ---
